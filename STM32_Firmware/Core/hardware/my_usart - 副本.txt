#include "my_usart.h" 
#include <string.h> // 用于 memset

// --- 外部变量声明 ---
// 告诉编译器，huart1和huart2这两个变量在别的文件里定义(main.c)
// 这样我们就可以在这里使用它们
extern UART_HandleTypeDef huart1;
extern UART_HandleTypeDef huart2;

// --- 模块私有定义和变量 ---
#define RX_BUFFER_SIZE 1   // 每次只接收一个字节
#define CMD_BUFFER_SIZE 9  // 完整的指令帧大小

// static关键字让这些变量只能在usart.c文件内部访问，对main.c不可见
static uint8_t rx_buffer[RX_BUFFER_SIZE];    // USART2接收缓冲区
static uint8_t cmd_buffer[CMD_BUFFER_SIZE];  // 存储一帧完整的指令
static volatile uint8_t cmd_received_flag = 0; // 指令接收完成标志

// --- 函数实现 ---

void UART_Init(void)
{
    // 启动USART2的中断接收，这是整个接收流程的起点
    HAL_UART_Receive_IT(&huart2, rx_buffer, RX_BUFFER_SIZE);
}

uint8_t Is_Command_Ready(void)
{
    return cmd_received_flag;
}

void Get_Received_Command(CommandData_t* cmd_data)
{
    if (cmd_received_flag == 1)
    {
        // 从内部缓冲区解析数据到用户提供的结构体中
        cmd_data->cmd_type  = cmd_buffer[1];
        cmd_data->channel   = cmd_buffer[2];
        cmd_data->wave_type = cmd_buffer[3];
        cmd_data->frequency = (cmd_buffer[4] << 8) | cmd_buffer[5];
        cmd_data->phase     = (cmd_buffer[6] << 8) | cmd_buffer[7];

        // 处理完毕，清除标志位
        cmd_received_flag = 0;
    }
}


void Send_Waveform_Command_To_FPGA(uint8_t channel, uint8_t wave_type, uint16_t frequency, uint16_t phase)
{
    uint8_t fpga_frame[CMD_BUFFER_SIZE];

    fpga_frame[0] = 0xA5; // 帧头
    fpga_frame[1] = 0x01; // 命令字: 设置波形
    fpga_frame[2] = channel;
    fpga_frame[3] = wave_type;
    fpga_frame[4] = (uint8_t)(frequency >> 8);
    fpga_frame[5] = (uint8_t)(frequency & 0xFF);
    fpga_frame[6] = (uint8_t)(phase >> 8);
    fpga_frame[7] = (uint8_t)(phase & 0xFF);
    fpga_frame[8] = 0x5A; // 帧尾

    HAL_UART_Transmit(&huart1, fpga_frame, CMD_BUFFER_SIZE, 100);
}

// --- 中断回调函数 ---
// HAL库会默认寻找这个函数，我们把它放在这里，它就自然地处理usart.c中的私有变量
void HAL_UART_RxCpltCallback(UART_HandleTypeDef *huart)
{
    if(huart->Instance == USART2)
    {
        static uint8_t rx_state = 0;
        static uint8_t byte_count = 0;
        uint8_t received_byte = rx_buffer[0];

        if (rx_state == 0)
        {
            if (received_byte == 0xA5)
            {
                cmd_buffer[0] = 0xA5;
                byte_count = 1;
                rx_state = 1;
            }
        }
        else if (rx_state == 1)
        {
            cmd_buffer[byte_count] = received_byte;
            byte_count++;

            if (byte_count >= CMD_BUFFER_SIZE)
            {
                if (cmd_buffer[CMD_BUFFER_SIZE - 1] == 0x5A)
                {
                    cmd_received_flag = 1; // 设置标志
					  HAL_GPIO_TogglePin(GPIOF, GPIO_PIN_9); 
                }
                rx_state = 0;
                byte_count = 0;
            }
        }
        HAL_UART_Receive_IT(&huart2, rx_buffer, RX_BUFFER_SIZE); // 重新启动接收
    }
}
